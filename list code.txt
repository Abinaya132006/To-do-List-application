<script>
let colorIndex = 1;

function addTask() {
    let input = document.getElementById("taskInput");
    let text = input.value.trim();
    if (!text) { alert("Enter a task"); return; }

    let li = document.createElement("li");

    // Assign soft colors in loop
    li.classList.add("task-color-" + colorIndex);
    colorIndex++;
    if (colorIndex > 5) colorIndex = 1;

    li.innerHTML = `
        <div class="task-box" onclick="toggleTask(this)">
            <input type="checkbox">
            <span>${text}</span>
        </div>
        <button class="delete-btn" onclick="deleteTask(this)">✕</button>
    `;

    document.getElementById("taskList").appendChild(li);
    input.value = "";
}

function toggleTask(box) {
    let li = box.parentElement;
    li.classList.toggle("completed");
    let check = box.querySelector("input");
    check.checked = !check.checked;
}

function deleteTask(btn) {
    btn.parentElement.remove();
}
</script>
